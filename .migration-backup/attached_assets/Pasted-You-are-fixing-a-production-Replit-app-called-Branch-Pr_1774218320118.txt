2. blendedRevenuePerDay
if averageLengthOfStayDays is less than or equal to 60, use rhcDay1To60
otherwise use:
((rhcDay1To60 times 60) plus (rhcDay61Plus times (averageLengthOfStayDays less 60))) divided by averageLengthOfStayDays

3. annualRevenue
equals targetADC times blendedRevenuePerDay times 365

4. annualVariableCost
equals targetADC times totalVariableCostPerDay times 365

5. annualPayroll
equals the sum of all annual staffing row costs returned by the staffing model

6. annualOverhead
equals monthlyNonPayrollOverhead times 12

7. annualFixedCost
equals annualPayroll plus annualOverhead

8. annualProfit
equals annualRevenue less annualVariableCost less annualPayroll less annualOverhead

9. operatingMarginPercent
equals annualProfit divided by annualRevenue times 100

10. contributionPerDay
equals blendedRevenuePerDay less totalVariableCostPerDay

11. breakEvenADC
equals annualFixedCost divided by (contributionPerDay times 365)

12. targetMarginADC
equals annualFixedCost divided by (365 times ((blendedRevenuePerDay times (1 less targetOperatingMarginPercent divided by 100)) less totalVariableCostPerDay))

13. monthlyAdmissionsNeeded
equals (targetADC times 365) divided by averageLengthOfStayDays divided by 12

14. weeklyAdmissionsNeeded

15. marketersNeededRaw

16. marketersNeededDisplay
